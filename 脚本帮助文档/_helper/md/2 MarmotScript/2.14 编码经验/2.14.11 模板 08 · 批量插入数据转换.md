```javascript
#!/usr/bin/env node
/**
 * Template 08: Batch Insert / Data Convert（批量操作 + 数据转换）
 *
 * 来源模式：生产脚本的批量写入场景
 * 真实代表：SPUC_SUBMIT_PO-8 / SLOD_EXEC_AFTER_CREATE_DELIVERY / SITF_POST_IMP_BATCH_SCUX_COMMON11 / SITF_REL_TABLE_RECORD_INVOKE-5
 *
 * 适用场景：
 *   - 大量数据导入（>100 行）
 *   - 多表关联插入（头+行）
 *   - 批量操作分块（chunk）
 *
 * 使用方法：
 *   1. 复制本文件
 *   2. 替换 TODO（表名 / 字段映射 / 分块大小）
 *   3. 测试：100 行 / 1000 行 / 10000 行
 */

const _ = require("lodash");
const objectOperation = require("@M/v1/object");
const moment = require("moment");

const logPrefix = "SRM-<CUSTOMER>|ADAPTOR|<TASK_CODE>";
const CHUNK_SIZE = 100;   // 推荐分块大小（v3 引擎优化值）

/**
 * 入口：处理批量导入
 * input 形如：
 *   {
 *     records: [ {header}, {header}, ... ]  // 业务单据数组
 *   }
 */
function process(input) {
  STD.Logger.info(`${logPrefix} 入参：${STD.SAFE_JSON.stringify(input)}`);

  if (_.isEmpty(input.records)) {
    return input;
  }

  let records = input.records;

  // 1. 构造待插入行（按业务规则转换）
  let insertLines = [];
  records.forEach(record => {
    let lines = buildLines(record);
    insertLines.push(...lines);
  });

  // 2. 分块插入（避免大批量性能/超时问题）
  if (insertLines.length > 0) {
    _.chunk(insertLines, CHUNK_SIZE).forEach(batch => {
      objectOperation.create({
        object: "<LINE_TABLE>",     // TODO: 如 sodr_po_line
        record: batch               // array of records
      });
    });
  }

  STD.Logger.info(`${logPrefix} 批量插入完成：${insertLines.length} 行`);
  return input;
}

/**
 * 业务转换：单条 record → 多行
 */
function buildLines(record) {
  let lines = [];

  record.items.forEach(item => {
    lines.push({
      // 主键外键
      <HEADER_ID>: record.<HEADER_ID>,   // TODO

      // 业务字段
      itemCode: item.code,
      itemName: item.name,
      quantity: item.qty,
      uomCode: item.uom,

      // 通用审计字段
      tenantId: STD.CurrentContext.getTenantId(),
      creationDate: moment().format("YYYY-MM-DD HH:mm:ss"),
      createdBy: STD.CurrentContext.getUserId(),
      enabledFlag: 1,

      // 扩展字段（attribute 系列）
      attributeVarchar1: item.attr1 || "",
      attributeDecimal1: item.dec1 || 0,
      attributeLongtext1: item.longtext1 ? JSON.stringify(item.longtext1) : ""
    });
  });

  return lines;
}

/**
 * 模式 B：头 + 行 一起插入（事务一致性）
 */
function insertHeaderAndLines(headerData, linesData) {
  // 1. 插入头
  let header = objectOperation.create({
    object: "<HEADER_TABLE>",
    record: headerData
  });
  let headerId = header.<HEADER_ID_FIELD>;  // TODO

  // 2. 给每行打上 headerId
  linesData.forEach(line => line.<HEADER_ID> = headerId);

  // 3. 分块插入行
  _.chunk(linesData, CHUNK_SIZE).forEach(batch => {
    objectOperation.create({
      object: "<LINE_TABLE>",
      record: batch
    });
  });

  return header;
}

/**
 * 模式 C：批量更新（白名单字段）
 */
function batchUpdate(records, fields) {
  if (_.isEmpty(records)) return;
  // 分块更新（避免大批量锁表）
  _.chunk(records, CHUNK_SIZE).forEach(batch => {
    objectOperation.updateOptional({
      object: "<TABLE>",
      record: batch,
      fields: fields
    });
  });
}

/**
 * 模式 D：批量删除
 */
function batchDelete(ids) {
  if (_.isEmpty(ids)) return;
  _.chunk(ids, CHUNK_SIZE).forEach(batch => {
    objectOperation.remove({
      object: "<TABLE>",
      id: batch
    });
  });
}

/**
 * 模式 E：批量按主键查（替代循环 selectById）
 */
function batchFetch(ids) {
  if (_.isEmpty(ids)) return [];
  // fetchById 支持 id 数组
  return objectOperation.fetchById({
    object: "<TABLE>",
    id: ids
  }) || [];
}
```
