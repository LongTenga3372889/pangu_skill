# 模板 07 · 编码规则生成
> summary: 编码规则生成模板（PO/SN/DN/批次号），serialGenerator.generateCode
> 引用路径: @经验库/模板/2.14.10

> 真实代表：SLOD_GENERATE_DELIVERY_NUM-1 / SLOD_GENERATE_DELIVERY_NUM / SPCM_PC_NUM_INIT-3
> 使用方法：复制下方代码骨架 → 全局替换 `<CUSTOMER>`/`<TASK_CODE>`/`<running_service>`/`<SYSTEM>` 占位符 → 替换业务字段（TODO 标记）→ 配置 Constants → 测试部署

```javascript
#!/usr/bin/env node
/**
 * Template 07: Code Rule Generator（编码规则生成）
 *
 * 来源模式：真实编码生成类脚本
 * 真实代表：SLOD_GENERATE_DELIVERY_NUM-1 / SLOD_GENERATE_DELIVERY_NUM / SPCM_PC_NUM_INIT-3
 *
 * 适用场景：
 *   - 单号/批次号/标签号自动生成
 *   - PO/SN/DN/条码 等业务单据编码
 *   - 带前缀/带变量映射的复合编码
 *
 * 使用方法：
 *   1. 在 srm-adaptor「编码规则」先配置 ruleCode（关联 sequence）
 *   2. 复制本文件
 *   3. 替换 TODO（ruleCode / 变量 / 返回 key）
 */

const objectOperation = require("@M/v1/object");
const serialGenerator = require("@M/v1/serial-generator");
const _ = require("lodash");

const logPrefix = "SRM-<CUSTOMER>|ADAPTOR|<TASK_CODE>";

/**
 * 入口：按节点模板代码分支
 * input 形如：
 *   {
 *     param1: header,          // 业务单据头
 *     param2: deliveryParamVO  // 参数 VO（含 nodeTemplateCode）
 *   }
 */
function process(input) {
  STD.Logger.info(`${logPrefix} 入参：${STD.SAFE_JSON.stringify(input)}`);

  let header = input.param1;
  let paramVO = input.param2;

  // 按业务类型分发
  if (paramVO && paramVO.nodeTemplateCode === "<TEMPLATE_CODE>" && header) {
    return generateAsnNum(header);
  }
  // 其他业务类型继续扩展
  // if (paramVO?.nodeTemplateCode === "OTHER") return generateOtherNum(header);

  return {};
}

/**
 * 模式 A：ASN 单号生成
 */
function generateAsnNum(header) {
  // 1. 取业务实体编码（OU）
  let ouCode = requireOuCode(header);
  if (!ouCode) {
    throw new STD.SimpleException("hpfm.error.ou_required");
  }

  // 2. 生成编码（带 OU 变量）
  let num = serialGenerator.generateCode({
    ruleCode: "<RULE_CODE>",       // TODO: 如 SLOD_PYLONTECH_ASN_NUM
    variableMap: {
      customSeq: ouCode            // 变量映射：sequence 会按 OU 分段
    }
  });

  // 3. 可选：加前缀
  let fullNum = "[X]" + num;        // TODO: 按业务约定

  STD.Logger.info(`${logPrefix} 生成单号：${fullNum}`);

  // 4. 返回标准结构
  return {
    result: {
      <KEY>: fullNum                // TODO: 如 deliveryNum
    }
  };
}

/**
 * 模式 B：批次号生成（多变量）
 */
function generateLotNum(supplierCode, itemCode) {
  let num = serialGenerator.generateCode({
    ruleCode: "<LOT_RULE_CODE>",    // TODO
    variableMap: {
      supplierCode: supplierCode,
      itemCode: itemCode
    }
  });
  return num;
}

/**
 * 取 OU 编码（带异常处理）
 */
function requireOuCode(header) {
  if (!header.ouId) return null;
  let ouInfo = objectOperation.fetchById({
    object: "hpfm_operation_unit",
    id: header.ouId
  });
  return ouInfo ? ouInfo.ouCode : null;
}

/**
 * 模式 C：基于供应商的编码（外部供应商 → 内部编码映射）
 */
function generateBySupplier(header) {
  let supplierCode = "";
  if (header.supplierId) {
    let supplier = objectOperation.fetchById({
      object: "sslm_external_supplier",
      id: header.supplierId
    });
    supplierCode = supplier?.supplierNum || "";
  } else if (header.supplierCompanyId) {
    let supplierCo = BIZ.CompanyHelper.companyInfoById(header.supplierCompanyId);
    supplierCode = supplierCo?.companyNum || "";
  }
  if (!supplierCode) {
    throw new STD.SimpleException("hpfm.error.supplier_required");
  }
  return "X" + serialGenerator.generateCode({
    ruleCode: "<RULE_CODE>",
    variableMap: {supplierCode: supplierCode}
  });
}

/**

 * □ variableMap 的 key 与编码规则配置一致
 * □ 测试：OU 缺失 / 供应商缺失 / 重复生成
```
