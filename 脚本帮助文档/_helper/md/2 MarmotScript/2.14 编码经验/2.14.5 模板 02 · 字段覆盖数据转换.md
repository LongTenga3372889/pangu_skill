#!/usr/bin/env node
/**
 * Template 02: Field Override / Data Convert（字段覆盖数据转换，轻量级）
 *
 * 来源模式：D2 报告模板 B
 * 真实代表：SLOD_UPDATE_DATA_BY_CREATE-5 / SSLM_LIFE_CYCLE_STAGE_CHANGE_APPROVR_AFTER-3 / SPUC_ORDER_SAVE_DATA_CONVERT-2
 *
 * 适用场景：
 *   - SPUC/SSLM/SSTA/SLOD/SPRM/_AFTER/_BEFORE/_PROCESS 钩子
 *   - 字段覆盖/数据转换（attributeXxx 字段写入）
 *   - 无外部调用 — 纯 ORM
 *
 * 使用方法：
 *   1. 复制本文件 → 新建脚本
 *   2. 替换 TODO 标记（表名 + 字段映射）
 *   3. 直接 return input（D2 主旋律）
 */

const objectOperation = require("@M/v1/object");
const _ = require("lodash");

// 日志前缀
const logPrefix = "SRM-<CUSTOMER>|ADAPTOR|<TASK_CODE>";
let tenantId = STD.CurrentContext.getExecutionTenantId();

/**
 * 入口函数
 * input 形如：
 *   {
 *     <HEADER 或 LINE DTO>,
 *     param1: ..., param2: ...
 *   }
 */
function process(input) {
  STD.Logger.debug(`${logPrefix} 入参：${STD.SAFE_JSON.stringify(input)}`);

  // 0. 防御性判断（避免陷阱 T02）
  let data = input.<DTO_FIELD>;  // TODO: 替换 DTO 字段名
  if (!data) return input;

  // 1. 读取参考数据（按需多表查询）
  let header = objectOperation.fetchById({
    object: "<HEADER_TABLE>",   // TODO: 如 sodr_po_header
    id: data.<HEADER_ID>         // TODO: 如 poHeaderId
  }) || {};

  let uom = objectOperation.fetchById({
    object: "smdm_uom",
    id: data.uomId
  }) || {};

  let inventory = objectOperation.fetchById({
    object: "hpfm_inventory",
    id: data.inventoryId
  }) || {};

  // 2. 计算新字段值
  let computedValue1 = computeSomething(header, uom);
  let computedValue2 = computeSomethingElse(header, inventory);

  // 3. in-place 改字段（修改 input 对象）
  data.attributeVarchar1 = uom.attributeVarchar1;        // TODO: 字段映射
  data.attributeVarchar2 = inventory.inventoryCode;
  data.attributeDecimal1 = computedValue1;
  data.attributeLongtext1 = JSON.stringify(computedValue2);

  // 4. 局部写回（fields 白名单，避免覆盖审计字段）
  if (data.<HEADER_ID>) {
    objectOperation.updateOptional({
      object: "<HEADER_TABLE>",
      record: header,
      fields: ["attributeVarchar1", "attributeDecimal1", "attributeLongtext1"]
      // ⚠️ 严禁省略 fields（陷阱：v3 updateOptional fields 必填）
    });
  }

  STD.Logger.debug(`${logPrefix} 出参：${STD.SAFE_JSON.stringify(input)}`);
  return input;
}

/**
 * 计算函数示例
 */
function computeSomething(header, uom) {
  // TODO: 业务计算逻辑
  return header.quantity * (uom.conversion || 1);
}

function computeSomethingElse(header, inventory) {
  // TODO
  return {poNum: header.poNum, invCode: inventory.inventoryCode};
}
