#!/usr/bin/env node
/**
 * Template 04: Workflow Trigger / Approval Callback（工作流触发审批回调）
 *
 * 来源模式：D2 报告模板 D
 * 真实代表：SSLM_EXTERNAL_SUPPLIER_APPROVE_AFTER / SPUC_PO_CLOSE_OR_CANCEL_SKIP_ADAPTOR / SINV_RCV_TRX_DATA_AFTER_PUBLISH
 *
 * 适用场景：
 *   - SSLM/SSRC/SPUC/SSTA 的 _APPROVE_AFTER / _APPROVE_PROCESS / _AFTER_APPROVE 钩子
 *   - 业务审批启动（供应商准入/订单关闭/收货审批）
 *   - 审批历史查询（workFlow.historyApproval）
 *
 * 使用方法：
 *   1. 复制本文件
 *   2. 配置 documentCode + categoryCode（已在 srm-workflow 配置）
 *   3. 替换 variables 字段映射
 */

const _ = require("lodash");
const workFlow = require("@M/v1/workflow");
const objectOperation = require("@M/v1/object");
const queryblock = require("@M/v1/query-block");
const employeeHelper = require("@M/v1/employee");
const context = require("@M/v1/context");
const {userId, tenantNum} = context.basic();   // 全局上下文
let logger = require("@M/v1/logger");

const logPrefix = "SRM-<CUSTOMER>|ADAPTOR|<TASK_CODE>";

function process(input) {
  logger.info({msg: `${logPrefix} input: {}`, strs: [JSONBig.stringify(input)]});
  triggerWorkflow(input);
  return input;
}

function triggerWorkflow(input) {
  // 0. 加载主单据
  let header = objectOperation.fetchById({
    object: "<HEADER_TABLE>",    // TODO: 如 sslm_external_supplier_req
    id: input.<HEADER_ID>          // TODO: 如 extRefId
  });
  if (!header || !header.<HEADER_NUM>) {
    return;  // 静默返回（避免无意义的审批启动）
  }

  // 1. 准备流程变量（按 workflow 配置的字段名）
  let variables = [];
  generateVariables(header, variables);

  // 2. 构造 processStartDTO
  let processStartDTO = {
    documentCode: "<DOCUMENT_CODE>",    // TODO: 如 SCUX.WATSONS.EXT_SUPPLIER_REQ1
    categoryCode: "<CATEGORY_CODE>",    // TODO: 如 SCUX.WATSONS.EXT_SUPPLIER_REQ
    businessKey: header.<HEADER_NUM> + "+<DOCUMENT_CODE>",
    variables: variables
    // 可选：
    // title: "审批标题",
    // callbackUrl: "...",
  };

  // 3. 启动工作流
  try {
    let processInstanceId = workFlow.start(processStartDTO);
    logger.info({msg: `${logPrefix} processInstanceId: {}`, strs: [processInstanceId]});
    // 4. （可选）写回 processInstanceId 到业务单据
    header.<PROCESS_INSTANCE_FIELD> = processInstanceId;  // TODO: 如 attributeProcessId
    objectOperation.updateOptional({
      object: "<HEADER_TABLE>",
      record: header,
      fields: ["<PROCESS_INSTANCE_FIELD>"]
    });
  } catch (e) {
    STD.Logger.error(`${logPrefix} 工作流启动失败：${e.stack || e.toString()}`);
    throw new STD.SimpleException("hpfm.error.workflow_start_failed");
  }
}

/**
 * 生成审批变量（按字段名 + 值）
 * ⚠️ 字段名必须在 workflow 模板中已定义
 */
function generateVariables(object, variables) {
  // TODO: 替换字段列表（按 workflow 模板）
  let fields = [
    'field1', 'field2', 'field3'   // 如 ['reqNumber', 'supplierName', 'area']
  ];
  fields.forEach(field => {
    let variable = {name: field, value: ""};
    if (object[field]) {
      variable.value = object[field];
    }
    variables.push(variable);
  });
}

/**
 * 可选：审批历史查询（用于回调场景）
 */
function queryApprovalHistory(businessKey) {
  let history = workFlow.historyApproval({
    businessKey: businessKey
  });
  logger.info({msg: `${logPrefix} history: {}`, strs: [JSONBig.stringify(history)]});
  return history;
}
