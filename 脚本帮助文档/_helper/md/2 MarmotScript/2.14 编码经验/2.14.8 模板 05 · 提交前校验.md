#!/usr/bin/env node
/**
 * Template 05: Validate Before Submit / Approval（审批前/提交前校验）
 *
 * 来源模式：D2 报告模板 E
 * 真实代表：SPUC_PO_SUBMIT_VAILDATE-10 / SSRC_RFX_CHECK_PRICE_SUBMIT_VALIDATE-5 / SSRC_RFX_OFF_LINE_QUOTATION_SUBMIT_VALIDATE
 *
 * 适用场景：
 *   - _VALIDATE / _BEFORE_PROCESS / _CHECK_BEFORE 钩子
 *   - 提交前/审批前/导入前的数据校验
 *   - 多行校验 + 错误聚合
 *
 * 使用方法：
 *   1. 复制本文件
 *   2. 替换 TODO（校验对象 + 校验规则）
 *   3. 错误聚合后一次性抛出
 */

const _ = require("lodash");
const objectOperation = require("@M/v1/object");

const logPrefix = "SRM-<CUSTOMER>|ADAPTOR|<TASK_CODE>";

function process(input) {
  STD.Logger.info(`${logPrefix} 入参：${STD.SAFE_JSON.stringify(input)}`);

  // 0. 防御性判断
  if (_.isEmpty(input.inData) || !input.inData.<HEADER_FIELD>) {
    return input;
  }

  let header = input.inData.<HEADER_FIELD>;  // TODO: 如 inData.poHeader
  let invalidLines = [];

  // 1. 加载关联数据
  let lines = objectOperation.query({
    object: "<LINE_TABLE>",        // TODO: 如 sodr_po_line
    condition: { <HEADER_ID>: header.<HEADER_ID> }  // TODO: 如 poHeaderId: header.poHeaderId
  });

  // 2. 逐行校验
  for (let line of lines) {
    let errors = validateLine(line, header);
    if (errors.length > 0) {
      invalidLines.push({lineNum: line.lineNum, errors: errors});
    }
  }

  // 3. 错误聚合：一次性抛出（避免重复打扰用户）
  if (invalidLines.length > 0) {
    let msg = invalidLines.map(l =>
      `行 ${l.lineNum}: ${l.errors.join(";")}`
    ).join("\n");
    throw msg;
    // 或用 SimpleException：
    // throw new STD.SimpleException("hpfm.error.po_amount_invalid");
  }

  // 4. 校验通过：返回 input
  STD.Logger.info(`${logPrefix} 校验通过`);
  return input;
}

/**
 * 单行校验规则（按业务需求扩展）
 */
function validateLine(line, header) {
  let errors = [];

  // 规则 1：金额必须为正数
  if (!line.amount || line.amount <= 0) {
    errors.push("金额必须大于0");
  }

  // 规则 2：交货日期不能早于今天
  if (line.needByDate && moment(line.needByDate).isBefore(moment(), "day")) {
    errors.push("交货日期不能早于今天");
  }

  // 规则 3：物料必须属于当前 OU
  if (line.invOrganizationId !== header.invOrganizationId) {
    errors.push("库存组织不一致");
  }

  // 规则 4：参考数据校验
  let itemInfo = objectOperation.fetchById({
    object: "smdm_item",
    id: line.itemId
  });
  if (!itemInfo) {
    errors.push("物料不存在");
  } else if (itemInfo.enabledFlag != 1) {
    errors.push("物料已停用");
  }

  // TODO: 按业务添加更多规则

  return errors;
}
