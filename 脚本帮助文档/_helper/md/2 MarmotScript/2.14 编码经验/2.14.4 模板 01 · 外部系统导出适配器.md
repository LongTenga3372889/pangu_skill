#!/usr/bin/env node
/**
 * Template 01: External System Export Adapter（外部系统导出适配器）

 *
 * 适用场景：
 *   - SITF_*_INVOKE / SITF_*_EXPORT / SITF_*_EXP 类型的适配器
 *   - 调用 SAP/金蝶/用友 NC/OA/钉钉/飞书 等外部系统的 HTTP 接口
 *   - 输入：inData 数组；输出：outData 数组（同步状态）

 * 使用方法：
 *   1. 复制本文件 → 新建脚本
 *   2. 替换模板中的 TODO 标记

// 顶部：必填字段（不可省略）
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

编码：<TASK_CODE>                // 例：SITF_PO_EXP_INVOKE

适配器-入参示例：                // 必填，方便后续维护

// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 顶部常量加载（必须用 STD.SAFE_JSON.parse，原生 JSON.parse 不安全，陷阱 T06）
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
let tenantId;
let externalSystemCode;

let itfCtx = STD.SAFE_JSON.parse(STD.Constants.get("SITF_<CUSTOMER>_<SYSTEM>_CONFIG") || "{}");

// 日志前缀（统一规范，便于 ELK 检索）
const logPrefix = "SRM-<CUSTOMER>|ADAPTOR|<TASK_CODE>";

// 客户名替换表（避免硬编码）
const CUSTOMER_NAME = "<CUSTOMER>";

// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 入口函数（v3 引擎唯一合法入口）
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
function process(input) {
  STD.Logger.info(`${logPrefix} 入参：${STD.SAFE_JSON.stringify(input)}`);

  // 0. 防御：入参合法性校验（避免陷阱 T02 链式访问崩溃）
  if (!input || !input.interfaceHeader || !Array.isArray(input.inData)) {
    throw new SimpleException("hpfm.error.adapter.input_invalid");
  }

  // 1. 提取租户和外部系统编码（interfaceHeader 必填字段）
  tenantId = input.interfaceHeader.relationId;
  externalSystemCode = input.interfaceHeader.externalSystemCode;

  // 2. 准备输出容器
  let inData = input.inData;
  let outData = (input.outData = []);

  // 3. 逐条处理：per-item try/catch（避免陷阱 T01 / T17）
  inData.forEach(data => {
    try {
      // 3.1 参数准备
      let requestData = buildRequest(data);
      // 3.2 调用外部系统
      let invokeRsp = invokeExternal(requestData);
      // 3.3 响应处理（成功/失败都 push 到 outData）
      outData.push(handleResponse(invokeRsp, data));
    } catch (e) {
      // 必须 catch 后 push 到 outData（不 throw 出 process！）
      STD.Logger.error(`${logPrefix} 调用失败：${e.stack || e.toString()}`);
      outData.push({
        // 按外部系统约定字段命名（一般是 businessId / processInstanceId / poNum 等）
        businessKey: data.businessKey || data.processInstanceId || data.poNum,
        syncStatus: "ERROR",
        syncResponseMsg: e.toString()
      });
    }
  });

  // 4. 返回 input（v3 主旋律：≈2500+ 脚本使用 return input）
  STD.Logger.info(`${logPrefix} 出参：${STD.SAFE_JSON.stringify(input)}`);
  return input;
}

// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 业务函数：构造请求报文（按外部系统格式）
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
function buildRequest(data) {
  // TODO: 按外部系统字段映射构造请求体
  // 示例：SAP 订单同步
  return {
    HEADER: {
      BUSNKEY: data.poNum,
      DATE: moment().format("YYYYMMDD"),
      TIME: moment().format("HHMMSS"),
      SYSID: itfCtx.systemId,
      TECHKEY: itfCtx.techKey
    },
    BODY: [{
      EBELN: data.displayPoNum,
      BSART: data.poTypeCode,
      LIFNR: data.supplierCode,
      // ... 其他字段
    }]
  };
}

// HTTP 调用（Bearer Token 模式，95% 集成场景）
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
function invokeExternal(requestData) {
  // 1. 取 token（带缓存：避免每条都调）
  let token = getOrFetchToken();

  // 2. 构造 header
  let httpHeaders = {
    Authorization: "Bearer " + token,
    "Content-Type": "application/json"
  };

  // 3. 调用主接口（TODO: 替换 url）
  let url = itfCtx.host + itfCtx.apiUrl;
  let invokeRsp = STD.HttpClient.doPost(url, httpHeaders, requestData);
  // ⚠️ 解析前确认有 timeout 配置（陷阱 T08）
  STD.Logger.info(`${logPrefix} invokeRsp：${invokeRsp}`);
  return STD.SAFE_JSON.parse(invokeRsp);
}

// Token 缓存（避免陷阱 T08 雪崩 + 重复调用）
let tokenCache = {token: null, expireAt: 0};
function getOrFetchToken() {
  if (tokenCache.token && Date.now() < tokenCache.expireAt) {
    return tokenCache.token;
  }
  // 调用取 token 接口
  let tokenUrl = itfCtx.host + itfCtx.tokenUrl;
  let tokenRsp = STD.HttpClient.doGet(tokenUrl, null, null);
  let tokenObj = STD.SAFE_JSON.parse(tokenRsp);
  if (!tokenObj || !tokenObj.access_token) {
    throw "获取 TOKEN 失败：" + tokenRsp;
  }
  // 缓存（提前 5 分钟过期，留 buffer）
  tokenCache = {
    token: tokenObj.access_token,
    expireAt: Date.now() + (tokenObj.expires_in - 300) * 1000
  };
  return tokenObj.access_token;
}

// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 响应处理（按外部系统 code 判定成功/失败，陷阱 T10）
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
function handleResponse(invokeRsp, data) {
  // TODO: 按外部系统的 success code 判定（参考 coding-patterns.md P10）
  //   SAP:    code == 200
  //   飞书:   code == 0
  //   钉钉:   errcode == 0
  //   金蝶:   operResult == "1"
  //   用友 NC: result.status == 200
  let isSuccess = invokeRsp.code === 200;
  return {
    businessKey: data.processInstanceId || data.poNum,
    externalBusinessId: invokeRsp.id || invokeRsp.messageId,
    syncStatus: isSuccess ? "SUCCESS" : "ERROR",
    syncResponseMsg: invokeRsp.message || STD.SAFE_JSON.stringify(invokeRsp)
  };
}

function invokeWithBasicAuth(requestData) {
  let basic = itfCtx.userName + ":" + itfCtx.password;
  let basicAuth = OTHER.ITF_TOOL.base64Encode(basic);

  return STD.SAFE_JSON.parse(invokeRsp);
}
