> 使用方法：复制下方代码骨架 → 全局替换 `<CUSTOMER>`/`<TASK_CODE>`/`<running_service>`/`<SYSTEM>` 占位符 → 替换业务字段（TODO 标记）→ 配置 Constants → 测试部署

```javascript
#!/usr/bin/env node
/**
 * Template 06: Attachment Upload/Download（附件上传下载链路）
 *
 * 来源模式：D2 报告模板 F
 * 真实代表：SITF_RFX_RELEASE_EXP_INVOKE / SITF_PC_CONTRACT_DATA_INVOKE-8 / SITF_SETTLE_PAYMENT_EXP_INVOKE-14 / SITF_RCV_ASYNC_EXP
 *
 * 适用场景：
 *   - 附件外发：SRM → 外部系统（SAP/金蝶/OA）
 *   - 附件下载：外部系统 → SRM
 *   - 文件元数据查询：UUID → 文件列表
 *
 * 使用方法：
 *   1. 复制本文件
 *   2. 替换 TODO（externalUrl / externalAuth / bucketName）
 *   3. 测试：空 UUID / 大文件 / 超时
 */

const _ = require("lodash");
const attachment = require("@M/v1/attachment");

const logPrefix = "SRM-<CUSTOMER>|ADAPTOR|<TASK_CODE>";

/**
 * 入口：处理附件外发
 */
function process(input) {
  STD.Logger.info(`${logPrefix} 入参：${STD.SAFE_JSON.stringify(input)}`);

  let inData = input.inData;
  inData.forEach(data => {
    try {
      // 1. 取 SRM 附件 UUID 对应的文件列表
      let files = getSRMFiles(data.attachmentUuid);
      if (_.isEmpty(files)) {
        throw "附件 UUID 未找到";
      }

      // 2. 推到外部系统（按外部系统 API 适配）
      let uploadResults = pushToExternal(files, data);

      // 3. 回填外部系统的文件 ID
      data.externalFileId = uploadResults.fileId;
      data.syncStatus = "SUCCESS";
    } catch (e) {
      STD.Logger.error(`${logPrefix} 附件外发失败：${e.stack || e.toString()}`);
      data.syncStatus = "ERROR";
      data.syncResponseMsg = e.toString();
    }
  });

  return input;
}

/**
 * 模式 A：取 SRM 附件文件列表
 */
function getSRMFiles(attachmentUuid) {
  if (!attachmentUuid) return [];
  return attachment.fetchServiceFilesByUUID({uuid: attachmentUuid});
}

/**
 * 模式 A：附件外发（SRM → 外部）
 */
function pushToExternal(files, data) {
  // 1. 取外部系统 token（按外部系统）
  let token = obtainExternalToken();

  // 2. 构造 fileInfos 数组
  let fileInfos = files.map(file => ({
    fileUrl: file.fileUrl,
    bucketName: "private-bucket",   // ⚠️ SRM 固定私有桶
    fileStreamName: "file",
    fileName: file.fileName
  }));

  // 3. 上传到外部
  let result = attachment.uploadAttachmentByUrl({
    url: itfCtx.uploadUrl,           // TODO: 替换为外部上传 URL
    fileInfos: fileInfos,
    headers: {
      Authorization: "Bearer " + token
    },
    body: {
      parent_type: itfCtx.parentType,  // TODO: 飞书/OA 等
      parent_node: itfCtx.parentNode,
      size: files[0].fileSize,
      file_name: files[0].fileName
    }
  });
  STD.Logger.info(`${logPrefix} 附件上传结果：${result}`);

  // 4. 解析响应（按外部系统 code 判定成功）
  let parsed = STD.SAFE_JSON.parse(result);
  if (!parsed?.data?.file_token) {
    throw "外部系统文件上传失败：" + result;
  }
  return {fileId: parsed.data.file_token};
}

/**
 * 模式 B：附件下载（外部 → SRM）
 */
function downloadFromExternal(externalFileUrl) {
  // 1. 下载外部文件到字节流
  let fileBytes = attachment.fetchBufferByFile({fileUrl: externalFileUrl});

  // 2. 生成新 UUID（关联到业务单据）
  let newUuid = attachment.getAttachmentUUID();

  // 3. （可选）上传到 SRM 并关联业务单据
  // 这里需要业务场景的 attachmentUuid（如 poHeader.attachmentUuid）
  // attachment.uploadToAttachment({...});

  return newUuid;
}

/**
 * 模式 C：临时签名 URL（外部系统访问 SRM 文件）
 */
function getSignedUrl(fileUrl, expiresSeconds) {
  return attachment.getSignedUrl({
    bucketName: "private-bucket",
    url: fileUrl,
    expires: expiresSeconds || 600   // 默认 10 分钟
  });
}

/**
 * Token 缓存
 */
let tokenCache = {token: null, expireAt: 0};
function obtainExternalToken() {
  if (tokenCache.token && Date.now() < tokenCache.expireAt) {
    return tokenCache.token;
  }
  let tokenBody = {
    app_id: itfCtx.appId,
    app_secret: itfCtx.appSecret
  };
  let tokenInfo = STD.HttpClient.doPost(itfCtx.tokenUrl, {"Content-Type": "application/json"}, tokenBody);
  let tokenObj = STD.SAFE_JSON.parse(tokenInfo);
  if (!tokenObj || !tokenObj.access_token) {
    throw "获取 token 失败：" + tokenInfo;
  }
  tokenCache = {
    token: tokenObj.access_token,
    expireAt: Date.now() + (tokenObj.expires_in - 300) * 1000
  };
  return tokenObj.access_token;
}

/**
 * Checklist:
 * □ 替换 itfCtx 的 tokenUrl / uploadUrl / appId / appSecret
 * □ bucketName 必须为 "private-bucket"
 * □ expires 不超过 3600 秒（外部系统限制）
 * □ 大文件（>10MB）建议 base64 + chunk
 * □ 测试：空 UUID / token 过期 / 网络中断
 */
```
