# ext-itf-convert 外部接口转换
> summary: 外部接口报文转换工具，支持请求/响应报文格式转换与自定义变量
> 引用路径: @M/v1/ext-itf-convert

外部接口报文转换工具

**引入**

```javascript
//引包
const {ExtItfConvert} = require("@M/v1/ext-itf-convert");
//或者 let ExtItfConvert = require("@M/v1/ext-itf-convert").ExtItfConvert;
//创建实例
let convertClient = new ExtItfConvert("配置编码", "转换类型"); //转换类型可不填，默认请求报文和反馈报文都查询

```

***

### dataConvert

数据转换

#### 入参

| **参数名**        | **必输** | **类型** | **默认值** | **说明** |
|----------------|--------|--------|---------|--------|
| convertType    | 是      | string | 空       | 转换类型   |
| object         | 是      | object | 空       | 源数据    |
| customVariable | 否      | object | 空       | 变量     |

#### 使用示例

```javascript
function process(input) {
    //引包
    const {ExtItfConvert} = require("@M/v1/ext-itf-convert");
    //创建实例
    let ExtItfConvertClient = new ExtItfConvert(input.extItfCode);
    let customVariable = input.interfaceHeader;
    //请求报文转换
    let requestData = ExtItfConvertClient.dataConvert({
        convertType: "request",
        customVariable: customVariable,
        object: input.requestData
    });
    //反馈报文转换
    let responseData = ExtItfConvertClient.dataConvert({
        convertType: "response",
        customVariable: customVariable,
        object: input.responseData
    });
    return {requestData: requestData, responseData: responseData};
}

```

### getConfig

获取配置

#### 使用示例

```javascript
function process(input) {
    //引包
    let {ExtItfConvert} = require("@M/v1/ext-itf-convert");
    //创建实例
    let ExtItfConvertClient = new ExtItfConvert(input.extItfCode);
    //请求报文转换
    let allConfig = ExtItfConvertClient.getConfig();
    return {allConfig: allConfig};
}

```

**返回结果格式示例**

```json
{
  "allConfig": {
    "debugFlag": true,
    "uniqueKey": null,
    "constants": {},
    "varList": [],
    "requestFieldInfo": {},
    "responseFieldInfo": {},
    "id2ObjVarMap":{},
    "code2FunctionMap":{},
    "code2ScriptMap": {}
  }
}
```

### setCustomVariable

读取外部接口定义变量配置，设置手工新建的自定义变量

#### 使用示例

```javascript
function process(input) {
    //引包
    let {ExtItfConvert} = require("@M/v1/ext-itf-convert");
    //创建实例
    let ExtItfConvertClient = new ExtItfConvert(input.extItfCode);
    let customVariable = {};
    //设置自定义变量
    ExtItfConvertClient.setCustomVariable(customVariable, input.requestData);
    return {customVariable: customVariable};
}

```
