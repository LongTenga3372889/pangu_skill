## Base64

#### 描述

*   多种 Base64 编码/解码方式，适用于不同的兼容需求。

#### 使用示例
```javascript
    let cryptToolBase64 = require("@M/v1/crypt").Crypt.Base64;
     // base64字符串编码,默认UTF-8
    let encryptFile = cryptToolBase64.encodeBase64URLSafeString(value, charset);
    // 将base64字符串按照oldCharset解码后按照newCharset编码
    let decryptFile = cryptToolBase64.decodeBase64(value, oldCharset, newCharset);
    // 将字符流byte[]编码,返回byte[]
    let decryptFile = cryptToolBase64.encodeBouncycastle(buffer);
    // 将字符流byte[]解码,返回byte[]
    let decryptFile = cryptToolBase64.decodeBouncycastle(buffer);
    // base64字符串编码
    let decryptFile = cryptToolBase64.encodeSafe(buffer, charset);
    // base64字符串编码
    let decryptFile = cryptToolBase64.decodeSafe(buffer, charset);
    STD.Logger.debug("decryptFile :{}", Buffer.from(decryptFile).toString());
```

## AES

### 描述

*   支持三种模式：CBC、CFB、ECB
* 1.支持CBC、CFB和ECB模式，用法与ECB类似，默认填充方式-PKCS5Padding
* 2.CBC模式和CFB模式中iv参数必传，ECB模式中iv无效  
* 3.若content为字符串，使用者请注意charset参数传值，需符合【java.lang.String#getBytes(java.lang.String)】方法支持的参数

### CBC 模式
```javascript
    let Crypt = require("@M/v1/crypt").Crypt;
    Crypt.AES.CBC.encrypt({ content, key, keyType, iv, ivType, charset })
    Crypt.AES.CBC.decrypt({ content, key, keyType, iv, ivType })
```
- `content`：可为字符串或二进制数据。
- `key`：加密密钥。
- `iv`：初始化向量。
- `keyType` / `ivType`：如 `"base64"`,可不传,默认"base64"。
- `charset`：可不传,默认 `"UTF-8"`。
### CFB 模式
```javascript
    let Crypt = require("@M/v1/crypt").Crypt;
    Crypt.AES.CFB.encrypt({ content, key, keyType, iv, ivType, charset })
    Crypt.AES.CFB.decrypt({ content, key, keyType, iv, ivType })
```
- 与 CBC 类似，仅加密模式不同。

### ECB 模式
```javascript
    let Crypt = require("@M/v1/crypt").Crypt;
    Crypt.AES.ECB.encrypt({ content, key, keyType, charset })
    Crypt.AES.ECB.decrypt({ content, key, keyType })

    let obj = {
        //SecretKey
        key: "ZHNhZHNhZHNhZHNhZHN3ZQ==",
        //可传值base64或者hex, 非必输， 默认base64
        keyType: "base64",
        //IvParameterSpec，ECB模式可不传
        // iv: "mOAr1K8sb7uTmpATAaO8oQ==",
        //可传值base64或者hex, 非必输， 默认base64
        // ivType: "base64",
        //buffer或者字符串
        content: "上海",
        //非必输，默认UTF-8 ---- 当content为字符串时有用
        charset: "utf-8",
    };
    //ECB模式的AES加密
    let encryptFile = cryptToolAES.ECB.encrypt(buffer, key);
    //ECB模式的AES解密
    let decryptFile = cryptToolAES.ECB.decrypt(encryptFile, key);
    STD.Logger.debug("decryptFile :{}", Buffer.from(decryptFile).toString());
```
- 无需 IV，适用于某些兼容性场景。

## SM4
* 只支持 ECB 模式：
* SM4的provider参数一定要填写
```javascript
    //SM4加解密, 当前仅支持ECB模式
    let cryptToolSM4 = require("@M/v1/crypt").Crypt.SM4;
    let param = {
        //buffer
        data:Buffer.from("上海"),
        //buffer
        key:Buffer.from("F83F7C1C50889F33E6745594F78EB47F", "hex"),
        provider:"BC"
    };
    let encryptSM4 = cryptToolSM4.ECB.encrypt(param);
    param.data = encryptSM4;
    let decryptSM4 = cryptToolSM4.ECB.decrypt(param);
    STD.Logger.debug("解密后数据 :{} ", Buffer.from(decryptSM4).toString());
```

## DES
- 对称加密，适合处理简单加密需求。
```javascript
    let Crypt = require("@M/v1/crypt").Crypt;
    Crypt.DES.encrypt({ content, key })
    Crypt.DES.decrypt({ content, key })
```

## RSA加密（MD5withRSA）
* MD5withRSA方式
```javascript
    let Crypt = require("@M/v1/crypt").Crypt;
    let content = Buffer.from("qqqqq");
    // 公钥加密,RSA 加密最大块，默认 117,charset默认"UTF-8"
    Crypt.RSA.MD5withRSA.encryptByPublicKey({ content, publicKey, charset, maxEncryptBlock })
    // 私钥解密 RSA 解密最大块，默认 128,charset默认"UTF-8"
    Crypt.RSA.MD5withRSA.decryptByPrivateKey({ content, privateKey, charset, maxDecryptBlock })
    // 内容签名 charset默认"UTF-8"
    Crypt.RSA.MD5withRSA.sign({ content, privateKey, charset })
    // 内容验签 charset默认"UTF-8"
    Crypt.RSA.MD5withRSA.verify({ content, publicKey, sign, charset })
```
