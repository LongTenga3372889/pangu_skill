# ding-tool 钉钉工具
> summary: 钉钉工具类，提供加解密和文件上传功能
> 引用路径: @M/v1/ding-tool

钉钉工具类

**引入**
```javascript
  let dingTool = require("@M/v1/ding-tool");
```

***
### Crypt
加解密

```javascript
    const cryptTool = require("@M/v1/ding-tool").Crypt;
    let aesToken = "";
    let aesKey = "";
    let corpId = "";
    let msg_signature = "";
    let timeStamp = "";
    let nonce = "";
    let encryptMsg = "";
    let crypto = { token: aesToken, encodingAesKey: aesKey, corpId: corpId };
    //解密---decrypt(msgSignature, timeStamp, nonce, encryptMsg)
    let decryptMsg = cryptTool.createCrypto(crypto).decrypt(
        msg_signature,
        timeStamp,
        nonce,
        encryptMsg
    );
    STD.Logger.debug("decryptMsg :"+ decryptMsg);
    //加密
    let successMap = cryptTool.createCrypto(crypto).encrypt("success");
    STD.Logger.debug("successMap : {}", successMap);
    //数字签名---signature(aesToken, timeStamp, nonce, encryptMsg)
    STD.Logger.debug("cryptTool : " + cryptTool.signature(aesToken, timeStamp, nonce, encryptMsg));
```

### uploadFile
上传文件

```javascript
    let dingTool = require("@M/v1/ding-tool");
    let request = {
        url:'',      //必输
        requestMethod:''     //必输
    };
    let header = {};
    let rs = dingTool.uploadFile(request, header, Buffer.from('你好呀!!!'));
```
