# csv-helper CSV工具
> summary: CSV工具类，支持生成和解析CSV文件
> 引用路径: @M/v1/csv-helper

csv工具类

**引入**
```javascript
  let csvHelper = require("@M/v1/csv-helper");
```

***
### generateCSV
生成单个CSV文件

#### 入参

| **参数名**    | **必输** | **类型** | **默认值** | **说明** |
| ---------- |--------|-------|---------|-------|
| header       | 是      | array | []      | 标题数组  |
| body | 是      | array | []      | 数据数组  |
| charset | 否      | string | UTF-8   | 字符集   |

#### 使用示例

```javascript
function process(input) {  
  let csvHelper = require("@M/v1/csv-helper")
  let head = ["第一列", "第二列", "第三列"];
  let body = [["1","2","3"],["2","3","4"]];
  //建议字符集传GBK，否则office可能会乱码
  let bytes = csvHelper.generateCSV(head, body, "GBK");
  return { result: bytes };
}

```



**返回结果示例**

```json
{
  //文件buffer，过长不宜展示
  "result" : [ -25, -84, -84, -28, -72, -128, -27, -120, -105, 44, -25......]
}


```

***
### parseFile
解析单个CSV文件

#### 入参

| **参数名**    | **必输** | **类型** | **默认值** | **说明**   |
| ---------- |--------|-------|---------|----------|
| buffer       | 是      | array | 无       | 文件buffer |
| charset | 否      | string | UTF-8   | 字符集      |

#### 使用示例

```javascript
function process(input) {
    let csvHelper = require("@M/v1/csv-helper");
    let head = ["第一列", "第二列", "第三列"];
    let body = [["1","2","3"],["2","3","4"]];
    //建议字符集传GBK，否则office可能会乱码
    let bytes = csvHelper.generateCSV(head, body, "GBK");
    let result = JSON.parse(csvHelper.parseFile(bytes, "GBK"));
    return { result: result };
}
```

**返回结果示例**

```json
{
  "result" : {
    "header" : [ "第一列", "第二列", "第三列" ],
    "content" : [ {
      "第一列" : "1",
      "第二列" : "2",
      "第三列" : "3"
    }, {
      "第一列" : "2",
      "第二列" : "3",
      "第三列" : "4"
    } ]
  }
}
```
