# SFTP客户端
> summary: SFTP文件传输客户端，支持文件上传、下载、删除操作
> 引用路径: @M/v1/sftp

***

# 引入

```javascript
const sftpClient = require("@M/v1/sftp");
```

# 使用案例

```javascript
function process(input) {
  //需要添加白名单，如192.168.1.1:22
  let connProp = {
    host: "10.213.116.121",
    port: "2222",
    username: "tester",
    password: "password",
    homePath: "./",
    sessionStrictHostKeyChecking: "no"
  };

  let fileParams = {
    directory: "test",
    fileName: "测试.txt"
  };

  //准备buffer(这东西理解成byte[]就行了)
  let buffer = Buffer.from('{"key": "value"}');

  let sftpClient = require("@M/v1/sftp");
  
  // 上传
  let result = sftpClient.uploadFile(connProp, {
    directory: "test",
    fileName: "测试.txt",
    buffer: buffer
  });

  STD.Logger.debug("result: {}", result);

  // 下载
  let fileBytes = sftpClient.downloadFile(connProp, {
    directory: "test",
    fileName: "测试.txt"
  });
  let fileContent = Buffer.from(fileBytes);
  STD.Logger.debug("fileContent: {}", fileContent);

  // 删除
  result = sftpClient.deleteFile(connProp, {
    directory: "test",
    fileName: "测试.txt"
  });
  STD.Logger.debug("result: {}", result);

  return {test:result};
}
```
