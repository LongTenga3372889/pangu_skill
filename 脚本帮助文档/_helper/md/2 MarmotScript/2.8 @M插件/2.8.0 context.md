# context 执行上下文
> summary: 执行时基本信息的获取，支持用户、租户、角色、时区、语言、环境等上下文查询
> 引用路径: @M/v1/context

执行时基本信息的获取（上下文中无用户信息取匿名用户）

**引入**

```javascript
let context = require("@M/v1/context");
```

***

### 基本上下文

```javascript
const context = require("@M/v1/context");
const {userId, tenantNum} = context.basic();
//返回例子
/*
{
    //用户id
    "userId" : "495568",
    //当前角色的租户id
    "tenantId" : "30",
    //当前角色的租户编码
    "tenantNum" : "SRM-ZHENYUN",
    //用户所属的租户id
    "organizationId" : "30",
    //当前角色id
    "roleId" : "1",
    //所有角色id列表
    "roleIds" : [ "1", "2", "52244", "52569", "52570" ],
    //用户时区
    "timeZone" : "GMT+8",
    //当前执行脚本的租户id
    "executionTenantId" : "30",
    //当前语言
    "language" : "zh_CN",
    //当前环境
    "env" : "DEV"
  }
  
 */

```
