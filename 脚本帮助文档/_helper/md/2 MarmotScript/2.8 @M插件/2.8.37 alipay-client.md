# alipay-client 支付宝服务端SDK
> summary: 支付宝服务端SDK工具，提供公钥模式接入，支持业务参数配置和请求发送
> 引用路径: @M/v1/alipay-client
支付宝服务端SDK工具

**引入**
```javascript
  let alipayClient = require("@M/v1/alipay-client");
```

***
### defaultAlipayClient
公钥模式接入SDK

```javascript
    let alipayClient = require("@M/v1/alipay-client");
    // 基础配置
    let alipayConfig = {
        // 网关地址
        serverUrl: "http://openapi.stable.dl.alipaydev.com/gateway.do",
        // appId 应⽤ID
        appId: "",
        // privateKey 应⽤私钥 
        privateKey: "",
        // alipayPublicKey ⽀付宝公钥
        alipayPublicKey: "",
        // 请求格式，固定值json
        format:"json",
        // 字符集
        charset:"GBK",
        // 签名类型
        signType:"RSA2"
    };
    // 业务参数对象
    let bizContentObj = {
        collect_id: "",
        product_code: "",
        serial_no: "",
        data_content: ""
    };
    // 请求参数
    let alipayRequest = {
        // 其他请求参数
        udfParams: {
            ws_service_url: ""
        },
        apiMethodName: "",
        // 业务参数
        bizContent: STD.SAFE_JSON.stringify(bizContentObj)
    };
    let result = alipayClient.defaultAlipayClient(alipayConfig, alipayRequest);
```

### 注意：
> 1.alipayConfig用来实例化客户端

> 2.bizContent用来设置业务请求入参

> 3.udfParams是其他的请求参数，如ws_service_url、app_auth_token等参数
