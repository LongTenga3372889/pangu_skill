# 对象操作器Object
> summary: 数据库和配置表对象操作器，支持query/create/update/remove等CRUD操作，自带事务和租户隔离
> 引用路径: @M/v1/object

>用于操作数据库或配置表对象  
>>对象操作器属于涉Database操作，单次操作超时时间为10秒，不可更改
> 
> 对象操作器自带事务，如果不想加事务可在每个方法的入参中加入  
>>tx:false 
> 
> 如query(object:'table',tx:false)
> 
> 注意：关闭事务将无法查询到未提交的数据,且无法保证错误时回滚，请慎重使用该功能；
> oauth服务的脚本不支持使用事务。

## 条件查询（query）

### 结构化查询

**描述**

*   根据条件查询指定对象的数据，自带租户数据隔离（默认会拼上当前脚本的租户id作为条件）

**名称**

*   query

**入参**

| **参数名**   | **必输** | **类型** | **默认值** | **说明**                       |
| --------- | ------ | ------ | ------- |------------------------------|
| object    | 是      | string | 空       | 操作对象名称                       |
| condition | 否      | 对象     | {}      | 查询条件，如{templateCode:"XXXXX"} |
| tx        | 否      | boolean   | true    | 加入事务                         |
| usingRead | 否      | boolean   | false   | 使用读库查询                       |


#### 使用示例

```javascript
function process(input) {
   let objectOperation = require("@M/v1/object");
  
  /*
  操作表 sada_script_template_favorite
     */
  //注:此处的condition若为空,则会查询该表在本租户下所有数据(相当于只有一个tenant_id作为查询条件)
  //单次查询量最多为10w条，condition内属性重复不生效
    let res = objectOperation.query({object:"sada_script_template_favorite",condition:{code:"dog"}})
   STD.Logger.info("query: {}",res);
   //若为空 会返回一个空数组[] 请直接用if(res.length)判断是否为空
   if(res.length){
     STD.Logger.info("description: {}","有");
   }
}


```

**返回结果示例**

```json
//返回结果为一个数组
[{"objectVersionNumber":"1","lastUpdatedBy":"-1","code":"dog","createdBy":"-1","lastUpdateDate":"2022-06-01 15:29:12","id":"2","creationDate":"2022-06-01 15:29:12","userId":"123","_fp":"32"}]

```

### QL
**Query还有一种QL写法，用于满足还是想写SQL的用户**   
**QL与Query能力完全一致，其本质是QL解析为condition后再执行，QL仅仅是语法糖，无功能增强**

#### QL使用示例

```javascript
const objectOperation = require("@M/v1/object");
function process(input) {
  const queryResult = objectOperation.QL('select * from scux_srm_zhenyun_zhenyun_personal_lunch_fee where fee>200');
  STD.Logger.info("query: {}",queryResult);
  return { result: queryResult};
}


```

## 分页查询（page）

**描述**

*   按照条件分页查询指定对象的数据，自带租户数据隔离（默认会拼上当前脚本的租户id作为条件）

**名称**

*   page

**入参**

| **参数名**     | **必输** | **类型** | **默认值**          | **说明** |
| ----------- | ------ | ------ | ---------------- | ------ |
| object      | 是      | string | 空                | 操作对象名称 |
| condition   | 否      | obj    | {}               | 查询条件   |
| pageRequest | 否      | obj    | {page:0,size:10} | 分页对象   |
| tx        | 否      | boolean   | true    | 加入事务            |
| usingRead | 否      | boolean   | false   | 使用读库查询         |

#### 使用示例

```javascript
function process(input) {
    let objectOperation = require("@M/v1/object");
  //注:此处的condition若为空,则会查询该表在本租户下所有数据(相当于只有一个tenant_id作为查询条件)
  //单次查询量最多为10w条，condition内属性重复不生效
   let resPage = objectOperation.page({object:"sada_script_template_favorite",pageRequest:{page:0,size:3},condition:{code:"123"}})
   STD.Logger.info("resPage: {}",resPage);
   //若为空 会返回一个分页的对象，里面的content为空，可以用if(resPage.empty)来判断  内容在content里
   if(!resPage.empty){
     STD.Logger.info("page: {}","有");
   }else{
     STD.Logger.info("page: {}","没有");
   }

}

```

### 分页排序
```javascript
//升序排列size为2，page为0的数据
object.page({
    object: "scux_srm_zhenyun_adapter_test",
    condition: { numberTest: [1,10,100]},
    pageRequest:{
      page:0,
      size:2,
      sort:[
        {"property":"numberTest",
         direction:"ASC"}       //必须大写，降序DESC
      ]}

});
```
**返回结果示例**

```json
//返回结果为一个page对象， 通过empty属性判断是否存在值
{"totalPages":"1","totalElements":"0","numberOfElements":"0","size":"3","number":"0","content":[],"empty":true}


```

### 查询额外操作(适用于query和page)
#### 等于 
可用于所有数据类型

**注意 2025.03.29 "containNull"属性 以及"="上prod** 
```javascript
object.query({
    object: "sada_adaptor_execute_collection",
    condition: {
        dataFlag: {opr:'=', v:'1',containNull:false}  // containNull  默认false
    }
});
// 等于两种简写方式
// 1.简写1
object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataFlag: {opr:'=', v:'1'}  // containNull  默认false
  }
});
//2.简写2
object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataFlag: 1 
  }
});
// 特殊用法

object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataFlag: {opr:'=', v:'1',containNull:true}  
  }
});
//其sql为: select * from sada_adaptor_execute_collection where (data_flag=1 or data_flag is null);


```


#### 大于
可用于number、date数据类型
```javascript
//查询所有dataFlag>1的数据
object.query({
    object: "sada_adaptor_execute_collection",
    condition: {
        dataFlag: {opr:'>', v:'1',containNull:false}  //>=同理
    }
});
// 简写
object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataFlag: {opr:'>', v:'1'}  //>=同理  containNull  默认false
  }
});
//containNull 为true
object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataFlag: {opr:'>', v:'1',containNull:false}  //>=同理 
  }
});
// 其sql 简写为 select * from sada_adaptor_execute_collection where (data_flag>1 or data_flag is null);
```
#### 小于
可用于number、date数据类型
```javascript
//查询所有dataFlag<1的数据
object.query({
    object: "sada_adaptor_execute_collection",
    condition: {
      dataFlag: {opr:'<', v:'1',containNull:false}    //<=同理
    }
});
// 简写
object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataFlag: {opr:'<', v:'1'}    //<=同理 containNull  默认false
  }
});
//containNull 为true
object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataFlag: {opr:'<', v:'1',containNull:true}   
  }
});
// 其sql 简写为 select * from sada_adaptor_execute_collection where (data_flag<1 or data_flag is null);







```
#### 不等于
可用于所有数据类型
```javascript
//查询所有dataText与给定v不等的数据，不包含null值
object.query({
    object: "sada_adaptor_execute_collection",
    condition: {
        dataText: {opr:'!=', v:'b5314cb3-b237-438f-9cc9-0f3b11',containNull:false} 
    }
});
// 可简写
object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataText: {opr:'!=', v:'b5314cb3-b237-438f-9cc9-0f3b11'}// containNull 默认false
  }
});
// 特殊用法
object.query({
  object: "sada_adaptor_execute_collection",
  condition: {
    dataText: {opr:'!=', v:'b5314cb3-b237-438f-9cc9-0f3b11',containNull:true}
  }

// 其执行sql为 select * from sada_adaptor_execute_collection where (data_text!=b5314cb3-b237-438f-9cc9-0f3b11 or data_text is null);

```
#### 在...之间
只能应用于number和date类型
```javascript
//查询numberTest数值在1 <= n <= 2之间的数据
result = object.query({
    object: "scux_srm_zhenyun_adapter_test",
    condition: { 
      numberTest: {opr:"between", v1:"1",v2:"2"}
    }
  });
```

```javascript
result = object.query({

    }
});
```

## 根据id查询数据（fetchById）

*   根据传入的id值查询指定对象的数据(注:id字段为固定字段,无需改为指定对象的主键字段)

*   fetchById

**入参**

| **参数名** | **必输** | **类型**  | **默认值** | **说明** |
| ------- | ------ | ------- | ------- | ------ |
| object  | 是      | string  | 空       | 操作对象名称 |
| id      | 是      | \[] 或 值 | 空       | 对象集合   |
| tx        | 否      | boolean   | true    | 加入事务    |
| usingRead | 否      | boolean   | false   | 使用读库查询  |

```javascript
function process(input) {
    let objectOperation = require("@M/v1/object");
```

## 更新指定字段（updateOptional）

**描述**

*   根据主键更新指定对象的指定字段（无需为query中查询出来的数据），单次最多更新10000行数据

**名称**

*   updateOptional

**入参**

| **参数名** | **必输** | **类型**  | **默认值** | **说明** |
| ------- | ------ | ------- | ------- | ------ |
| object  | 是      | string  | 空       | 操作对象名称 |
| record | 是      | \[]或者{} | 空       | 指定数据   |
| fields  | 是      | \[]或者值  | 空       | 指定更新字段 |

#### 使用示例

```javascript
function process(input) {
    let objectOperation = require("@M/v1/object");
    let fetchOneByIdForUpdateOptional = objectOperation.fetchById({object:"sada_script_template_favorite",id:3})
    STD.Logger.info("before --- updateOptional: {}",fetchOneByIdForUpdateOptional);
    fetchOneByIdForUpdateOptional.code = "bigDog"
    fetchOneByIdForUpdateOptional.userId = 778899
   //这里有两个值做了更改 但是未指定更新字段 则不会更新
```

## 计数（count）

**描述**

*   可根据条件计数

**名称**

*   count

**入参**


| **参数名**   | **必输** | **类型** | **默认值** | **说明**                       |
| --------- | ------ | ------ | ------- | ---------------------------- |
| object    | 是      | string | 空       | 操作对象名称                       |
| condition | 否      | 对象     | {}      | 查询条件，如{templateCode:"XXXXX"} |
| tx        | 否      | boolean   | true    | 加入事务                         |
| usingRead | 否      | boolean   | false   | 使用读库查询                       |

#### 使用示例

```javascript

function process(input) {
    let tableName = "hpfm_lov";
    let count = objectOperation.count({
        object: tableName,
        condition: { lovTypeCode: "SQL" }
    });
    //count 为查询结果的数量
    return { count: count };
}
```
## 对象拷贝（objectCopy）

**描述**

*   将数据拷贝至标准对象中，返回此对象数据集。

**名称**

*   objectCopy

**入参**

| **参数名** | **必输** | **类型** | **默认值** | **说明** |
| ------- | ------ | ------ | ------- |--------|
| object  | 是      | string | 空       | 操作对象名称 |
| record | 是      | []或者{} | 空       | 源数据    |

#### 使用示例

```javascript
function process(input) {
    let objectOperation = require("@M/v1/object");

    let data = objectOperation.fetchById({
        object: "hpfm_lov",
        id: ["10", "11"]
    });
    //冗余字段，测试用下
    data[0].test1 = "test1";
    data[0].test2 = "test2";
    data[0].test3 = "test3";
    STD.Logger.info("data: {}", data);

    //将data中的数据 拷贝至目标对象上
    let target = objectOperation.objectCopy({
        object: "hpfm_lov",
        record: data
    });
    return { data: target };
}
```
