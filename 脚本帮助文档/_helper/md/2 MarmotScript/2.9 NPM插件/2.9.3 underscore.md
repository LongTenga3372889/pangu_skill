## 引入

```javascript
const _=require('underscore');
```

## 获取对象的属性key值
```javascript
let keyArray = _.keys({one: 1, two: 2, three: 3});
// => ["one", "two", "three"]
```

## 过滤对象中的空属性
```javascript
let a = {
    a:null,
    b:1,
    c:2
}
//返回一个a被过滤掉的对象
let afterFilter = _.pick(a, (v,k,b)=>{
    //空属性会返回为false
    return !_.isNull(v);
});
//=> {b:1,c:2}
```

## 集合分组
```javascript
//根据整数部分分组
_.groupBy([1.3, 2.1, 2.4], function(num){ return Math.floor(num); });
//=> {1: [1.3], 2: [2.1, 2.4]}

//根据属性分组
_.groupBy(['one', 'two', 'three'], 'length');
//=> {3: ["one", "two"], 5: ["three"]}
```

## 集合排序
```javascript
let list = [{
    a:1,
    b:2
}];
//根据b，升序
let asc = _.sortBy(list, function (e){ return e.b});
//根据b，降序
let dec = _.sortBy(list, function (e){ return e.b}).reverse();
```
## 集合查找
```javascript
//查找并返回第一个值
let even = _.find([1, 2, 3, 4, 5, 6], function(num){ return num % 2 == 0; });
// => 2
```

## 生成对应属性的映射map
```javascript
let list = [
    {
        tenantNum:"SRM",
        tenantId:0
    },
    {
        tenantNum:"SRM-ZHENYUN",
        tenantId:30
    }
];
//根据tenantId生成租户信息的映射map
let result = _.indexBy(list, 'tenantId');
/*
* {
*   "0" : {tenantNum:"SRM",tenantId:0},
*   "30" : {tenantNum:"SRM-ZHENYUN",tenantId:30}
* }
* */
```
